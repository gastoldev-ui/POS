### `backend/src/routes/posRoutes.js`
```js
// backend/src/routes/posRoutes.js
import { Router } from "express";
import { imprimirTicket } from "../services/printerService.js";
import { guardarVenta, leerVentasDelDia } from "../services/posService.js";
import { verSiguienteTicket } from "../services/contadorService.js";

const router = Router();

// Helper: normaliza los pagos de una venta a un array [{metodo, monto}]
// Soporta el contrato nuevo (pagos[]) y el viejo (metodoPago + total).
function normalizarPagos(venta) {
  if (Array.isArray(venta.pagos) && venta.pagos.length > 0) {
    return venta.pagos.map((p) => ({
      metodo: String(p.metodo || "Efectivo"),
      monto: Number(p.monto) || 0,
    }));
  }
  return [
    {
      metodo: venta.metodoPago || "Efectivo",
      monto: Number(venta.total) || 0,
    },
  ];
}

// Valida un "YYYY-MM-DD" y devuelve el string o null.
function parsearFechaQuery(q) {
  if (typeof q !== "string") return null;
  return /^\d{4}-\d{2}-\d{2}$/.test(q) ? q : null;
}

// ==================== GET /api/pos/siguiente-ticket ====================
// Devuelve el próximo número (formateado) SIN reservarlo.
// Query opcional: ?fecha=YYYY-MM-DD (default: hoy).
router.get("/siguiente-ticket", (req, res) => {
  try {
    const fecha = parsearFechaQuery(req.query.fecha) || undefined;
    const id = verSiguienteTicket(fecha);
    res.json({ success: true, id });
  } catch (e) {
    res.status(500).json({ success: false, error: e.message });
  }
});

// ==================== POST /api/pos/venta ====================
// Guarda la venta en el JSON del día y dispara la impresión.
// El ID secuencial lo asigna el backend atómicamente al guardar.
router.post("/venta", async (req, res) => {
  try {
    const venta = req.body;

    if (!venta || !Array.isArray(venta.items) || venta.items.length === 0) {
      return res
        .status(400)
        .json({ success: false, error: "La venta no tiene ítems" });
    }
    if (!Number.isFinite(Number(venta.total)) || Number(venta.total) <= 0) {
      return res
        .status(400)
        .json({ success: false, error: "El total de la venta es inválido" });
    }

    // Si el cliente manda fecha en formato YYYY-MM-DD, respetamos ese
    // archivo; si no, el service usa venta.fecha / hoy.
    const fechaQuery = parsearFechaQuery(venta.fechaArchivo);
    const guardada = guardarVenta(venta, fechaQuery);

    const skipPrint = req.get("X-Skip-Print") === "1";
    if (skipPrint) {
      return res.json({
        success: true,
        venta: guardada,
        impresion: { success: true, skipped: true },
      });
    }

    let impresion;
    try {
      impresion = await imprimirTicket(guardada);
    } catch (e) {
      impresion = { success: false, error: e.message };
    }

    res.json({ success: true, venta: guardada, impresion });
  } catch (e) {
    console.error("❌ [POS/venta]", e.message);
    res.status(500).json({ success: false, error: e.message });
  }
});

// ==================== POST /api/pos/imprimir ====================
router.post("/imprimir", async (req, res) => {
  try {
    const resultado = await imprimirTicket(req.body);
    if (resultado.success) {
      res.json({ success: true });
    } else {
      res.status(500).json({ success: false, error: resultado.error });
    }
  } catch (e) {
    res.status(500).json({ success: false, error: e.message });
  }
});

// ==================== POST /api/pos/cerrar-caja ====================
router.post("/cerrar-caja", (req, res) => {
  try {
    const { montoInicial, efectivoDeclarado } = req.body || {};
    const ventas = leerVentasDelDia();

    const totalVentas = ventas.reduce(
      (sum, v) => sum + (Number(v.total) || 0),
      0,
    );

    const totalEfectivo = ventas.reduce((sum, v) => {
      const pagos = normalizarPagos(v);
      return (
        sum +
        pagos
          .filter((p) => p.metodo === "Efectivo")
          .reduce((s, p) => s + p.monto, 0)
      );
    }, 0);

    const cantidadVentasEfectivo = ventas.filter((v) =>
      normalizarPagos(v).some((p) => p.metodo === "Efectivo"),
    ).length;

    const inicial = Number(montoInicial) || 0;
    const declarado = Number(efectivoDeclarado) || 0;
    const esperadoEnCaja = inicial + totalEfectivo;
    const diferencia = declarado - esperadoEnCaja;

    res.json({
      success: true,
      resumen: {
        fecha: new Date().toISOString().slice(0, 10),
        cantidadVentas: ventas.length,
        cantidadVentasEfectivo,
        totalVentas,
        totalEfectivo,
        montoInicial: inicial,
        esperadoEnCaja,
        efectivoDeclarado: declarado,
        diferencia,
      },
    });
  } catch (e) {
    console.error("❌ [POS/cerrar-caja]", e.message);
    res.status(500).json({ success: false, error: e.message });
  }
});

// ==================== GET /api/pos/ventas-hoy ====================
// Query opcional: ?fecha=YYYY-MM-DD (default: hoy).
router.get("/ventas-hoy", (req, res) => {
  try {
    const fecha = parsearFechaQuery(req.query.fecha);
    const ventas = leerVentasDelDia(fecha || new Date());
    res.json({ success: true, total: ventas.length, ventas, fecha: fecha || null });
  } catch (e) {
    res.status(500).json({ success: false, error: e.message });
  }
});

// ==================== GET /api/pos/resumen-dia ====================
// Query opcional: ?fecha=YYYY-MM-DD (default: hoy).
// Devuelve el agregado del día: total, cantidad, desglose por método.
router.get("/resumen-dia", (req, res) => {
  try {
    const fecha = parsearFechaQuery(req.query.fecha);
    const ventas = leerVentasDelDia(fecha || new Date());

    const porMetodo = {};
    let totalFacturado = 0;

    for (const v of ventas) {
      const pagos = normalizarPagos(v);
      for (const p of pagos) {
        porMetodo[p.metodo] = (porMetodo[p.metodo] || 0) + p.monto;
        totalFacturado += p.monto;
      }
    }

    res.json({
      success: true,
      fecha: fecha || new Date().toISOString().slice(0, 10),
      cantidadVentas: ventas.length,
      totalFacturado,
      porMetodo,
    });
  } catch (e) {
    res.status(500).json({ success: false, error: e.message });
  }
});

export default router;```

### `backend/src/services/contadorService.js`
```js
// backend/src/services/contadorService.js
import fs from 'fs';
import path from 'path';
import { fileURLToPath } from 'url';

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);

const DATA_DIR = path.resolve(__dirname, '../../data/pos');
const CONTADOR_PATH = path.join(DATA_DIR, 'contador.json');

const PREFIJO = 'T-';
const DIGITOS = 5;

function fechaHoy() {
  return new Date().toISOString().slice(0, 10); // YYYY-MM-DD
}

function asegurarDir() {
  if (!fs.existsSync(DATA_DIR)) {
    fs.mkdirSync(DATA_DIR, { recursive: true });
  }
}

function leerContadores() {
  asegurarDir();
  if (!fs.existsSync(CONTADOR_PATH)) return {};
  try {
    return JSON.parse(fs.readFileSync(CONTADOR_PATH, 'utf-8'));
  } catch {
    // archivo corrupto → arrancamos de cero
    return {};
  }
}

function escribirContadores(obj) {
  asegurarDir();
  fs.writeFileSync(CONTADOR_PATH, JSON.stringify(obj, null, 2), 'utf-8');
}

/**
 * Formatea un número entero a "T-00001".
 */
function formatear(n) {
  return `${PREFIJO}${String(n).padStart(DIGITOS, '0')}`;
}

/**
 * Devuelve el PRÓXIMO número (formateado) sin reservarlo.
 * Útil para mostrar en la UI antes de cobrar.
 */
export function verSiguienteTicket(fecha = fechaHoy()) {
  const contadores = leerContadores();
  const actual = Number(contadores[fecha]) || 0;
  return formatear(actual + 1);
}

/**
 * Incrementa y RESERVA el siguiente número para la fecha dada.
 * Devuelve el ID formateado. Es atómico a nivel de proceso (Node
 * single-thread, pero igual escribimos antes de devolver para que
 * un crash no deje el contador desfasado del archivo de ventas).
 *
 * IMPORTANTE: una vez reservado, el número NO se reutiliza aunque
 * la impresión falle. La venta ya se persistió con ese ID.
 */
export function reservarSiguienteTicket(fecha = fechaHoy()) {
  const contadores = leerContadores();
  const actual = Number(contadores[fecha]) || 0;
  const nuevo = actual + 1;
  contadores[fecha] = nuevo;
  escribirContadores(contadores);
  return formatear(nuevo);
}```

### `backend/src/services/posService.js`
```js
// backend/src/services/posService.js
import fs from 'fs';
import path from 'path';
import { fileURLToPath } from 'url';
import { reservarSiguienteTicket } from './contadorService.js';

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);

const VENTAS_DIR = path.resolve(__dirname, '../../data/pos/ventas');

function obtenerRutaArchivo(fecha = new Date()) {
  if (!fs.existsSync(VENTAS_DIR)) {
    fs.mkdirSync(VENTAS_DIR, { recursive: true });
  }
  const fechaStr = fecha.toISOString().slice(0, 10); // YYYY-MM-DD
  return path.join(VENTAS_DIR, `${fechaStr}.json`);
}

/**
 * Parsea "YYYY-MM-DD" a un Date en UTC a mediodía, para que no se
 * desplace por el timezone al hacer toISOString() de vuelta.
 */
function parsearFechaStr(fechaStr) {
  if (!fechaStr) return new Date();
  const m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(fechaStr);
  if (!m) return new Date();
  const [, y, mo, d] = m;
  return new Date(Date.UTC(Number(y), Number(mo) - 1, Number(d), 12, 0, 0));
}

/**
 * Guarda una venta en el archivo del día.
 * Asigna el ID secuencial atómicamente ANTES de escribir, si la venta
 * no viene con uno ya asignado.
 *
 * @param {Object} venta
 * @param {string} [fechaStr] - YYYY-MM-DD. Si no viene, usa la fecha
 *                              interna de la venta (o hoy).
 */
export function guardarVenta(venta, fechaStr = null) {
  const fecha = fechaStr ? parsearFechaStr(fechaStr) : new Date(venta.fecha || Date.now());
  const fechaArchivo = fecha.toISOString().slice(0, 10);
  const ruta = obtenerRutaArchivo(fecha);

  let ventasDelDia = [];
  if (fs.existsSync(ruta)) {
    try {
      ventasDelDia = JSON.parse(fs.readFileSync(ruta, 'utf-8'));
    } catch {
      /* archivo corrupto, empezamos de cero */
    }
  }

  // El backend SIEMPRE asigna el ID secuencial (ignora el que mande el
  // front). Así garantizamos unicidad y orden.
  const idAsignado = reservarSiguienteTicket(fechaArchivo);
  const ventaFinal = { ...venta, id: idAsignado };

  ventasDelDia.push(ventaFinal);
  fs.writeFileSync(ruta, JSON.stringify(ventasDelDia, null, 2), 'utf-8');
  return ventaFinal;
}

/**
 * Lee las ventas del día.
 * @param {Date|string} [fecha=new Date()] - Date o "YYYY-MM-DD".
 */
export function leerVentasDelDia(fecha = new Date()) {
  const fechaDate = typeof fecha === 'string' ? parsearFechaStr(fecha) : fecha;
  const ruta = obtenerRutaArchivo(fechaDate);
  if (!fs.existsSync(ruta)) return [];
  try {
    return JSON.parse(fs.readFileSync(ruta, 'utf-8'));
  } catch {
    return [];
  }
}```

### `backend/src/services/printerService.js`
```js
// backend/src/services/printerService.js
import { printer as ThermalPrinter, types as PrinterTypes } from 'node-thermal-printer';

/**
 * Normaliza los pagos de un ticket a un array [{metodo, monto}].
 * Soporta el contrato nuevo (pagos[]) y el viejo (metodoPago + total).
 */
function normalizarPagos(ticketData) {
  if (Array.isArray(ticketData.pagos) && ticketData.pagos.length > 0) {
    return ticketData.pagos.map((p) => ({
      metodo: String(p.metodo || 'Efectivo'),
      monto: Number(p.monto) || 0,
    }));
  }
  return [
    {
      metodo: ticketData.metodoPago || 'Efectivo',
      monto: Number(ticketData.total) || 0,
    },
  ];
}

/**
 * Genera e imprime un ticket usando ESC/POS.
 * @param {Object} ticketData - Datos del ticket (items, total, etc.)
 * @returns {Promise<{success: boolean, skipped?: boolean, error?: string}>}
 */
export async function imprimirTicket(ticketData) {
  try {
    // ── Guard de desarrollo ──────────────────────────────────────────
    // Si no hay PRINTER_INTERFACE seteada, no intentamos instanciar el
    // driver: node-thermal-printer tira "No driver set!" y ensucia los
    // logs. Devolvemos skipped=true para que el front sepa que no se
    // imprimió, pero la venta YA está guardada por el caller.
    if (!process.env.PRINTER_INTERFACE) {
      console.log(
        '🖨️  [printer] PRINTER_INTERFACE no seteada → skip impresión (modo dev)',
      );
      return {
        success: true,
        skipped: true,
        error: 'Impresora no configurada (modo dev)',
      };
    }

    // La construcción va DENTRO del try para que un fallo del driver
    // no tire abajo el proceso del backend (evita el 502).
    const printer = new ThermalPrinter({
      type: PrinterTypes.EPSON, // o STAR, depende de tu impresora
      interface: process.env.PRINTER_INTERFACE,
      characterSet: 'PC850_MULTILINGUAL', // Para acentos y ñ
      removeSpecialCharacters: false,
      lineCharacter: '-',
    });

    const isConnected = await printer.isPrinterConnected();
    if (!isConnected) {
      return { success: false, error: 'Impresora no conectada' };
    }

    // --- Formato del ticket ---
    printer.alignCenter();
    printer.bold(true);
    printer.println('DULCE REINO');
    printer.bold(false);
    printer.println('Ticket de Venta');
    printer.drawLine();

    printer.alignLeft();
    printer.println(`Fecha: ${new Date(ticketData.fecha).toLocaleString('es-AR')}`);
    printer.println(`Ticket #: ${ticketData.id}`);
    printer.drawLine();

    // Items
    for (const item of ticketData.items) {
      printer.println(item.nombre);
      const lineaCantidadPrecio = `${item.cantidad} x $${item.precioUnitario.toFixed(2)}`;
      printer.leftRight(lineaCantidadPrecio, `$${item.subtotal.toFixed(2)}`);
    }

    printer.drawLine();
    printer.bold(true);
    printer.leftRight('TOTAL', `$${ticketData.total.toFixed(2)}`);
    printer.bold(false);

    // --- Pagos (soporta uno o varios medios) ---
    const pagos = normalizarPagos(ticketData);
    printer.drawLine();
    printer.println('PAGOS:');
    for (const p of pagos) {
      printer.leftRight(p.metodo, `$${p.monto.toFixed(2)}`);
    }

    // Vuelto: solo si lo pagado supera el total
    const totalPagado = pagos.reduce((s, p) => s + p.monto, 0);
    const vuelto = Math.max(0, totalPagado - Number(ticketData.total || 0));
    if (vuelto > 0.001) {
      printer.leftRight('Vuelto:', `$${vuelto.toFixed(2)}`);
    }

    printer.drawLine();
    printer.alignCenter();
    printer.println('¡Gracias por su compra!');
    printer.newLine();
    printer.newLine();
    printer.cut();

    await printer.execute();
    return { success: true };
  } catch (error) {
    console.error('Error al imprimir:', error);
    return { success: false, error: error.message };
  }
}```

set -euo pipefail

  for nom in "${EXCLUIR_NOMBRES[@]}"; do

    ! -name "dump-*.txt" \
    ! -name "*.log" \
    ! -name "*.min.js" \
    ! -name "*.min.css" \
    | sort | while read -r file; do
      rel="${file#./}"
      ext="${file##*.}"
      lineas_total=$(wc -l < "$file" 2>/dev/null || echo 0)

      echo "### \`${rel}\`"
      echo '```'"${ext}"

      if [ "$lineas_total" -gt "$MAX_LINEAS" ]; then
        head -n "$MAX_LINEAS" "$file"
        echo ""
        echo "... [TRUNCADO: ${lineas_total} líneas totales, mostrando ${MAX_LINEAS}] ..."
      else
        cat "$file"
      fi

      echo '```'
      echo ""
    done
} > "$OUTPUT_FILE"

echo "✅ Dump generado en: $OUTPUT_FILE"
echo "   Líneas: $(wc -l < "$OUTPUT_FILE")"
echo "   Tamaño: $(du -h "$OUTPUT_FILE" | cut -f1)"```

// src/api/client.js
const BASE = "/api"; // Vite lo redirige a http://localhost:4000 vía proxy

export class ApiError extends Error {
  constructor(message, status, body) {
    super(message);
    this.name = "ApiError";
    this.status = status;
    this.body = body;
  }
}

async function manejarRespuesta(res) {
  const contentType = res.headers.get("content-type") || "";
  const esJson = contentType.includes("application/json");
  const body = esJson
    ? await res.json().catch(() => null)
    : await res.text().catch(() => null);

  if (!res.ok) {
    const msg =
      (body && body.error) || (body && body.message) || `HTTP ${res.status}`;
    throw new ApiError(msg, res.status, body);
  }
  return body;
}

async function post(path, body, opts = {}) {
  const headers = { "Content-Type": "application/json" };
  if (opts.imprimir === false) {
    headers["X-Skip-Print"] = "1";
  }
  const res = await fetch(`${BASE}${path}`, {
    method: "POST",
    headers,
    body: JSON.stringify(body ?? {}),
  });
  return manejarRespuesta(res);
}

async function get(path) {
  const res = await fetch(`${BASE}${path}`);
  return manejarRespuesta(res);
}

function conFecha(path, fecha) {
  if (!fecha) return path;
  const sep = path.includes("?") ? "&" : "?";
  return `${path}${sep}fecha=${encodeURIComponent(fecha)}`;
}

export const api = {
  posVenta: (ticket, opts) => post("/pos/venta", ticket, opts),
  posCerrarCaja: (montoInicial, efectivoDeclarado) =>
    post("/pos/cerrar-caja", { montoInicial, efectivoDeclarado }),
  posVentasHoy: (fecha) => get(conFecha("/pos/ventas-hoy", fecha)),
  posResumenDia: (fecha) => get(conFecha("/pos/resumen-dia", fecha)),
  posSiguienteTicket: (fecha) => get(conFecha("/pos/siguiente-ticket", fecha)),

import { useState, useEffect, useRef, useMemo, useCallback } from "react";
import { usePOS } from "../context/POSContext";
import { api } from "../api/client";
import { formatearPrecio } from "../utils/formateo";

const METODOS = [
  { key: "F5", nombre: "Efectivo", pideMonto: true, color: "emerald" },
  { key: "F6", nombre: "Cta. DNI", pideMonto: false, color: "indigo" },
  { key: "F7", nombre: "Débito", pideMonto: false, color: "sky" },
  { key: "F8", nombre: "Transferencia", pideMonto: false, color: "violet" },
];

const METODO_POR_TECLA = Object.fromEntries(METODOS.map((m) => [m.key, m]));
const METODO_DEFAULT = METODOS[0];

function redondear2(n) {
  return Math.round((Number(n) || 0) * 100) / 100;
}

// Bloquea el keyup residual de una tecla que ya se manejó en keydown.
// Se usa al cerrar el modal para que el Enter de "confirmar cobro"
// no se filtre al POS de atrás.
function bloquearKeyupResidual() {
  const stop = (e) => {
    e.preventDefault();
    e.stopPropagation();
  };
  window.addEventListener("keyup", stop, true);
  window.addEventListener("keypress", stop, true);
  setTimeout(() => {
    window.removeEventListener("keyup", stop, true);
    window.removeEventListener("keypress", stop, true);
  }, 100);
}

export default function CobroModal({ onCerrar }) {
  const pos = usePOS();

  const [pagos, setPagos] = useState([]);
  const [metodoActivo, setMetodoActivo] = useState(null);
  const [montoTipeado, setMontoTipeado] = useState("");
  // Fases: 'cobro' | 'confirmarPago' | 'elegirImpresion' | 'procesando' | 'exito'
  const [fase, setFase] = useState("cobro");
  const [error, setError] = useState(null);
  // Selección del cartel de impresión: 0 = Imprimir (default), 1 = No imprimir
  const [opcionImpresion, setOpcionImpresion] = useState(0);
  const [resultadoExito, setResultadoExito] = useState(null);

  const inputMontoRef = useRef(null);

  const total = pos.totales.total;
  const totalPagado = useMemo(
    () => redondear2(pagos.reduce((s, p) => s + p.monto, 0)),
    [pagos],
  );
  const restante = useMemo(
    () => redondear2(total - totalPagado),
    [total, totalPagado],
  );
  const estaCubierto = restante <= 0.001;

  // Foco en el input de monto. Se dispara al elegir método (el input
  // recién ahí se monta). Usamos requestAnimationFrame para garantizar
  // que el DOM ya esté pintado, y un segundo intento por si otro
  // setTimeout (ej. el `enfocarPrecio` de PanelPOS) nos pisa el foco.
  useEffect(() => {
    if (!metodoActivo) return;
    const enfocar = () => {
      inputMontoRef.current?.focus();
      inputMontoRef.current?.select();
    };
    const raf = requestAnimationFrame(enfocar);
    const t = setTimeout(enfocar, 80); // reintento tras el setTimeout(50) de PanelPOS
    return () => {
      cancelAnimationFrame(raf);
      clearTimeout(t);
    };
  }, [metodoActivo]);

  // Auto-cierre del paso 'exito'
  useEffect(() => {
    if (fase !== "exito") return;
    const t = setTimeout(() => onCerrar(), 1800);
    return () => clearTimeout(t);
  }, [fase, onCerrar]);

  // -------------------- Acciones --------------------

  const seleccionarMetodo = useCallback(
    (metodo) => {
      if (fase !== "cobro") return;
      setError(null);

      if (!metodo.pideMonto) {
        if (restante <= 0) return;
        const nuevosPagos = [
          ...pagos,
          { metodo: metodo.nombre, monto: restante },
        ];
        setPagos(nuevosPagos);
        setMetodoActivo(null);
        setMontoTipeado("");
        const totalPagadoNuevo = redondear2(
          nuevosPagos.reduce((s, p) => s + p.monto, 0),
        );
        if (totalPagadoNuevo >= total - 0.001) {
          setFase("confirmarPago");
        }
        return;
      }

      setMetodoActivo(metodo);
      setMontoTipeado("");
    },
    [fase, restante, pagos, total],
  );

  const confirmarMonto = useCallback(() => {
    if (!metodoActivo) return;

    // Si el campo está vacío, asumimos el restante (que es lo que muestra el placeholder).
    // Esto permite cobrar el total con un solo Enter tras elegir Efectivo.
    const montoTipeadoReal =
      montoTipeado === "" ? restante : Number(montoTipeado);
    const n = Number(montoTipeadoReal);

    if (!Number.isFinite(n) || n <= 0) {
      setError("Ingresá un monto válido");
      return;
    }
    // Efectivo: se permite sobrepago (el vuelto se muestra en confirmarPago).
    // Otros métodos: no pueden superar el restante.
    if (metodoActivo.nombre !== "Efectivo" && n > restante + 0.001) {
      setError(`El monto supera el restante (${formatearPrecio(restante)})`);
      return;
    }

    const nuevoPago = { metodo: metodoActivo.nombre, monto: redondear2(n) };
    const nuevosPagos = [...pagos, nuevoPago];
    const totalPagadoNuevo = redondear2(
      nuevosPagos.reduce((s, p) => s + p.monto, 0),
    );
    const cubierto = totalPagadoNuevo >= total - 0.001;

    setPagos(nuevosPagos);
    setMetodoActivo(null);
    setMontoTipeado("");
    setError(null);

    if (cubierto) {
      setFase("confirmarPago");
    }
  }, [metodoActivo, montoTipeado, restante, pagos, total]);

  const quitarPago = useCallback(
    (idx) => {
      if (fase !== "cobro") return;
      setPagos((prev) => prev.filter((_, i) => i !== idx));
      setError(null);
    },
    [fase],
  );

  const volverACobro = useCallback(() => {
    setFase("cobro");
    setError(null);
  }, []);

  const irAConfirmarPago = useCallback(() => {
    if (!estaCubierto) {
      setError(`Falta cubrir ${formatearPrecio(restante)}`);
      return;
    }
    setFase("confirmarPago");
  }, [estaCubierto, restante]);

  const irAElegirImpresion = useCallback(() => {
    setOpcionImpresion(0); // default: imprimir
    setFase("elegirImpresion");
    setError(null);
  }, []);

  // -------------------- Confirmación final --------------------

  const confirmarCobro = useCallback(
    async (debeImprimir) => {
      setFase("procesando");
      setError(null);
      try {
        const ticket = pos.construirTicket(pagos);
        const data = await api.posVenta(ticket, { imprimir: debeImprimir });

        // Guardamos el resultado para mostrarlo en la pantalla de éxito
        setResultadoExito({
          debeImprimir,
          impresion: data.impresion || null,
          total: ticket.total,
        });
        setFase("exito");
        // El vaciado del carrito se hace recién al cerrar (o podés hacerlo acá)
        pos.vaciar();
        // Evitamos que el keyup/keypress de la tecla que disparó esto
        // se escape al POS de atrás.
        bloquearKeyupResidual();
      } catch (err) {
        setError(err.message);
        setFase("cobro");
      }
    },
    [pagos, pos],
  );

  // -------------------- Teclado --------------------

  useEffect(() => {
    const handler = (e) => {
      // Bloquear teclas del navegador
      if (["F5", "F6", "F7", "F8"].includes(e.key)) {
        e.preventDefault();
        e.stopPropagation();
      }

      // --- Fase éxito: Enter/click cierra; auto-cierre por timeout ---
      if (fase === "exito") {
        if (e.key === "Enter" || e.key === "Escape") {
          e.preventDefault();
          e.stopPropagation();
          bloquearKeyupResidual();
          onCerrar();
        }
        return;
      }

      // --- Fase procesando: ignorar teclas ---
      if (fase === "procesando") return;

      // --- Fase elegir impresión: ←/→ alternan, Enter confirma, Esc vuelve ---
      if (fase === "elegirImpresion") {
        if (e.key === "ArrowLeft" || e.key === "ArrowRight") {
          e.preventDefault();
          e.stopPropagation();

        if (e.key === "Enter") {

        if (e.key === "Escape") {

          e.stopPropagation();
          setFase("confirmarPago");
          return;
        }
        return;
      }

      // --- Fase confirmar pago ---
      if (fase === "confirmarPago") {
        if (e.key === "Escape") {
          e.preventDefault();
          e.stopPropagation();
          volverACobro();
          return;
        }
        if (e.key === "Enter") {
          e.preventDefault();
          e.stopPropagation();
          irAElegirImpresion();
          return;
        }
        return;
      }

      // --- Fase cobro ---
      if (fase !== "cobro") return;

      // 1. Atajos de métodos (F5-F8). Va PRIMERO para que funcionen
      // incluso si ya hay un método activo (ej. cambias de Efectivo a Débito).
      if (METODO_POR_TECLA[e.key]) {
        e.preventDefault();
        e.stopPropagation();
        seleccionarMetodo(METODO_POR_TECLA[e.key]);
        return;
      }

      // 2. Si hay un método activo (estamos tipeando el monto)

          e.preventDefault();
          setMetodoActivo(null);
          setMontoTipeado("");
          setError(null);
          return;

          e.preventDefault();
          e.stopPropagation();
          confirmarMonto();
          return;
        }

        const enInput = document.activeElement === inputMontoRef.current;
        if (!enInput && /^[0-9]$/.test(e.key)) {
          e.preventDefault();
          e.stopPropagation();
          inputMontoRef.current?.focus();
          setMontoTipeado((prev) => prev + e.key);
          return;
        }
        return;
      }

      // 3. Si NO hay método activo (navegando por los botones)
      if (e.key === "Enter") {
        e.preventDefault();
        e.stopPropagation();
        if (pagos.length === 0) {

  return (
    <div className="bg-slate-50 border border-slate-200 rounded-xl p-3">

              <span className="font-bold text-slate-800">

                  onClick={() => onQuitar(i)}

  if (Array.isArray(v.pagos) && v.pagos.length > 0) {

// Matchea una venta contra el filtro de método seleccionado.

    const pagos = normalizarPagos(venta);
    const metodos = new Set(pagos.map((p) => p.metodo));
    return metodos.size > 1 || venta.metodoPago === "Mixto";
  }
  return normalizarPagos(venta).some((p) => p.metodo === filtro);
}

  // -------------------- Teclado (UN SOLO listener) --------------------
  // Fase capture + stopPropagation, igual que antes, pero ahora ramifica
  // según `fase`. No creamos listeners paralelos.

  useEffect(() => {
    const handler = (e) => {
      // ===== FASE PREVIEW =====
      if (fase === "preview") {
        if (e.key === "Escape") {
          e.preventDefault();
          e.stopPropagation();
          volverALista();
          return;
        }
        if (e.key === "Enter") {
          e.preventDefault();
          e.stopPropagation();
          imprimirPreview();
          return;
        }
        return;
      }

      // ===== FASE LISTA =====
      if (e.key === "Escape") {
        e.preventDefault();
        e.stopPropagation();
        onCerrar();
        return;
      }

      // Si el foco está en el input de fecha, dejamos que el navegador
      // maneje flechas/Enter/espacio nativos (date picker). Solo Esc
      // cierra el modal (ya manejado arriba).
      if (document.activeElement === inputFechaRef.current) return;

      if (ventasFiltradas.length === 0) return;

      if (e.key === "ArrowDown") {
        e.preventDefault();
        e.stopPropagation();
        setIndiceSel((prev) => (prev + 1) % ventasFiltradas.length);
      } else if (e.key === "ArrowUp") {
        e.preventDefault();
        e.stopPropagation();
        setIndiceSel(
          (prev) =>
            (prev - 1 + ventasFiltradas.length) % ventasFiltradas.length,
        );
      } else if (e.key === "Enter") {
        e.preventDefault();
        e.stopPropagation();
        abrirPreview(ventasFiltradas[indiceSel]);
      }
    };

    window.addEventListener("keydown", handler, true);
    return () => window.removeEventListener("keydown", handler, true);
  }, [fase, ventasFiltradas, indiceSel, onCerrar, abrirPreview, volverALista, imprimirPreview]);

  // -------------------- Helpers de render --------------------

  const horaDe = (v) =>
    new Date(v.fecha).toLocaleTimeString("es-AR", {
      hour: "2-digit",
      minute: "2-digit",
    });

  // -------------------- RENDER: PREVIEW --------------------

  if (fase === "preview" && ventaPreview) {
    const pagos = normalizarPagos(ventaPreview);
    const totalPagado = pagos.reduce((s, p) => s + p.monto, 0);

      0,
      Math.round((totalPagado - Number(ventaPreview.total || 0)) * 100) / 100,
    );

    return (
      <div className="fixed inset-0 z-[100] flex items-center justify-center bg-black/60 p-4">
        <div className="bg-white rounded-2xl shadow-2xl w-full max-w-md overflow-hidden flex flex-col max-h-[90vh]">
          {/* HEADER */}
          <div className="bg-indigo-600 p-4 flex items-center justify-between flex-shrink-0">
            <div>
              <h3 className="text-xl font-black text-white">Vista previa</h3>
              <p className="text-indigo-100 text-xs font-mono">
                Ticket #{ventaPreview.id}
              </p>
            </div>
            <button
              onClick={volverALista}
              className="text-white hover:text-red-200 font-bold text-xl px-2"
              title="Volver (Esc)"
            >
              ✕
            </button>
          </div>

          {/* CUERPO tipo ticket */}
          <div className="flex-1 overflow-y-auto p-5 font-mono text-sm bg-slate-50">
            <div className="text-center mb-2">
              <p className="font-black text-base tracking-wide">DULCE REINO</p>
              <p className="text-xs text-slate-500">Ticket de Venta</p>
            </div>
            <div className="border-t border-dashed border-slate-300 my-2" />
            <div className="flex justify-between text-xs text-slate-600">
              <span>Fecha:</span>
              <span>
                {new Date(ventaPreview.fecha).toLocaleString("es-AR")}
              </span>
            </div>
            <div className="flex justify-between text-xs text-slate-600">
              <span>Ticket #:</span>
              <span>{ventaPreview.id}</span>
            </div>
            <div className="border-t border-dashed border-slate-300 my-2" />

            {/* Items */}
            <ul className="space-y-1">
              {(ventaPreview.items || []).map((it, i) => (
                <li key={i}>
                  <p className="text-slate-800">{it.nombre}</p>
                  <div className="flex justify-between text-xs text-slate-600">
                    <span>
                      {it.cantidad} x {formatearPrecio(it.precioUnitario)}
                    </span>
                    <span>{formatearPrecio(it.subtotal)}</span>
                  </div>
                </li>
              ))}
            </ul>

            <div className="border-t border-dashed border-slate-300 my-2" />
            <div className="flex justify-between font-black text-base text-slate-800">
              <span>TOTAL</span>
              <span>{formatearPrecio(ventaPreview.total)}</span>
            </div>

            <div className="border-t border-dashed border-slate-300 my-2" />
            <p className="text-xs text-slate-500 mb-1">PAGOS:</p>
            <ul className="space-y-0.5">
              {pagos.map((p, i) => (
                <li key={i} className="flex justify-between text-xs">
                  <span>{p.metodo}</span>
                  <span>{formatearPrecio(p.monto)}</span>
                </li>
              ))}
            </ul>
            {vuelto > 0.001 && (
              <div className="flex justify-between text-xs mt-1 font-semibold text-sky-700">
                <span>Vuelto:</span>
                <span>{formatearPrecio(vuelto)}</span>
              </div>
            )}

            <div className="border-t border-dashed border-slate-300 my-3" />
            <p className="text-center text-xs text-slate-500">
              ¡Gracias por su compra!
            </p>
          </div>

          {/* MENSAJES */}
          {msjExito && (
            <div className="px-4 py-2 bg-emerald-50 border-t border-emerald-200 text-emerald-700 text-sm font-bold text-center">
              {msjExito}
            </div>
          )}
          {error && (
            <div className="px-4 py-2 bg-red-50 border-t border-red-200 text-red-700 text-sm text-center">
              ⚠️ {error}
            </div>
          )}

          {/* FOOTER */}
          <div className="p-4 bg-white border-t border-slate-200 flex gap-2 flex-shrink-0">
            <button
              onClick={volverALista}
              className="flex-1 px-4 py-3 text-slate-600 font-bold bg-slate-100 hover:bg-slate-200 rounded-xl transition"
            >
              ← Volver
              <span className="block text-[10px] font-mono opacity-70 mt-0.5">
                (Esc)
              </span>
            </button>
            <button
              onClick={imprimirPreview}
              disabled={imprimiendo}
              className="flex-[2] bg-indigo-600 hover:bg-indigo-700 disabled:bg-slate-300 text-white font-black text-base py-3 rounded-xl transition shadow-lg flex items-center justify-center gap-2"
            >
              🖨️ {imprimiendo ? "Imprimiendo..." : "Imprimir"}
              <span className="text-[10px] font-mono opacity-80">(Enter)</span>
            </button>
          </div>
        </div>
      </div>
    );
  }

  // -------------------- RENDER: LISTA --------------------

  return (
    <div className="fixed inset-0 z-[100] flex items-center justify-center bg-black/60 p-4">
      <div className="bg-white rounded-2xl shadow-2xl w-full max-w-3xl overflow-hidden flex flex-col max-h-[90vh]">
        {/* HEADER */}
        <div className="bg-indigo-600 p-4 flex items-center justify-between">
          <div>
            <h3 className="text-2xl font-black text-white">Historial</h3>
            <p className="text-indigo-100 text-xs">
              Fecha: {fecha} · Enter abre vista previa
            </p>
          </div>
          <button
            onClick={onCerrar}
            className="text-white hover:text-red-200 font-bold text-xl px-2"
            title="Cerrar (Esc)"
          >
            ✕
          </button>
        </div>

        {/* CONTROLES: fecha + búsqueda + filtro */}
        <div className="p-4 border-b border-slate-200 bg-slate-50 space-y-2">
          <div className="flex gap-2 items-center">
            <label className="text-xs font-bold text-slate-600 uppercase whitespace-nowrap">
              Fecha
            </label>
            <input
              ref={inputFechaRef}
              type="date"
              value={fecha}
              max={FECHA_HOY()}
              onChange={(e) => setFecha(e.target.value || FECHA_HOY())}

            className="w-full text-base border-2 border-indigo-300 rounded-lg px-3 py-2 bg-white focus:outline-none focus:ring-2 focus:ring-indigo-500 focus:border-indigo-500"
          />

          <div className="flex flex-wrap gap-1.5">
            {FILTROS.map((f) => {
              const activo = filtroMetodo === f.key;
              return (
                <button
                  key={f.key}

                  );

              </tbody>
            </table>
          )}
        </div>

        {/* RESUMEN DEL DÍA (fijo en el footer) */}
        {resumen && (
          <div className="px-4 py-3 bg-indigo-50 border-t border-indigo-200">
            <div className="flex justify-between items-start gap-4 text-sm">
              <div className="flex-shrink-0">

                <p className="text-xl font-black text-indigo-800">

// src/components/PanelPOS.jsx
import { useState, useEffect, useRef } from "react";
import { usePOS } from "../context/POSContext";
import { formatearPrecio } from "../utils/formateo";
import { CATALOGO_MOCK } from "../mocks/catalogoMock";
import { usePOSAtajos } from "../hooks/usePOSAtajos";
import { usePOSBusqueda } from "../hooks/usePOSBusqueda";
import { usePOSCarga } from "../hooks/usePOSCarga";
import CobroModal from "./CobroModal";
import ArqueoCajaModal from "./ArqueoCajaModal";
import HistorialModal from "./HistorialModal";

// Orden de navegación con ← → en la barra superior
const CAMPOS_ORDEN = ["producto", "cantidad", "precio"];

export default function PanelPOS() {
  const pos = usePOS();

  // Modales
  const [modalCobroAbierto, setModalCobroAbierto] = useState(false);
  const [modalArqueoAbierto, setModalArqueoAbierto] = useState(false);
  const [modalHistorialAbierto, setModalHistorialAbierto] = useState(false);

  // Refs para foco
  const inputProductoRef = useRef(null);
  const inputCantidadRef = useRef(null);
  const inputPrecioRef = useRef(null);
  const grillaRef = useRef(null);

  // Flag para ignorar el Enter residual que viene del modal de cobro/arqueo
  const enterBloqueadoRef = useRef(false);

  // Hooks de lógica
  const busqueda = usePOSBusqueda(CATALOGO_MOCK);
  const carga = usePOSCarga({
    onProducto: (art, cantidad, precio) => {
      const filaVacia = pos.items.find((it) => it.vacia);
      if (filaVacia) {
        pos.agregarProducto(filaVacia.id, art, cantidad, precio);
      }
    },
    onManual: (precio, descripcion, cantidad) => {
      pos.agregarManual(precio, descripcion, cantidad); // <-- Pasamos la cantidad
    },
  });
  // Foco inicial en precio (carga rápida tipo teclado)
  useEffect(() => {
    const t = setTimeout(() => {
      inputPrecioRef.current?.focus();
      inputPrecioRef.current?.select();
    }, 100);
    return () => clearTimeout(t);
  }, []);

  // Auto-scroll al fondo: la última fila siempre visible.
  const ultimaFilaId = pos.items[pos.items.length - 1]?.id;
  useEffect(() => {
    const el = grillaRef.current;
    if (!el) return;
    const raf = requestAnimationFrame(() => {
      el.scrollTop = el.scrollHeight;
    });
    return () => cancelAnimationFrame(raf);
  }, [ultimaFilaId, pos.totales.cantidadItems]);

  // -------------------- Navegación ← → entre campos --------------------

  const navegarCampo = (origen, direccion) => {
    const idx = CAMPOS_ORDEN.indexOf(origen);
    if (idx === -1) return;
    const destinoIdx = Math.min(
      CAMPOS_ORDEN.length - 1,
      Math.max(0, idx + direccion),
    );
    const destino = CAMPOS_ORDEN[destinoIdx];

    if (destino === "producto") {
      inputProductoRef.current?.focus();
      inputProductoRef.current?.select();
    } else if (destino === "cantidad") {
      inputCantidadRef.current?.focus();
      inputCantidadRef.current?.select();
    } else if (destino === "precio") {
      inputPrecioRef.current?.focus();
      inputPrecioRef.current?.select();
    }
  };

  // Devuelve true si consumió la tecla (para que el caller haga return)
  const handleFlechasCampo = (origen, e) => {
    if (e.key === "ArrowLeft") {
      e.preventDefault();
      e.stopPropagation();
      navegarCampo(origen, -1);
      return true;
    }
    if (e.key === "ArrowRight") {
      e.preventDefault();
      e.stopPropagation();
      navegarCampo(origen, +1);
      return true;
    }
    return false;
  };

  // -------------------- Acciones reusables --------------------

  const enfocarProducto = () => {
    inputProductoRef.current?.focus();
    inputProductoRef.current?.select();
  };

  const enfocarCantidad = () => {
    if (modalCobroAbierto || modalArqueoAbierto || modalHistorialAbierto)
      return;
    inputCantidadRef.current?.focus();
    inputCantidadRef.current?.select();
  };

  const enfocarPrecio = () => {
    if (modalCobroAbierto || modalArqueoAbierto || modalHistorialAbierto)
      return;
    inputPrecioRef.current?.focus();
    inputPrecioRef.current?.select();
  };

  const abrirArqueo = () => setModalArqueoAbierto(true);
  const abrirHistorial = () => setModalHistorialAbierto(true);

  const borrarUltima = () => pos.quitarUltima();

  const cobrar = () => {
    if (pos.totales.cantidadItems > 0) setModalCobroAbierto(true);
  };

  const cerrarModalCobro = () => {
    setModalCobroAbierto(false);

    setTimeout(() => {
      enterBloqueadoRef.current = false;
    }, 150);

    // Devuelve el foco a Precio después de que el modal se desmonte
    setTimeout(() => {
      inputPrecioRef.current?.focus();
      inputPrecioRef.current?.select();
    }, 200); // 200ms da tiempo a React a limpiar el modal y a CobroModal a soltar el foco
  };

  const cerrarModalArqueo = () => {
    setModalArqueoAbierto(false);
    enterBloqueadoRef.current = true;
    setTimeout(() => {
      enterBloqueadoRef.current = false;
    }, 150);
  };

  const cerrarModalHistorial = () => {
    setModalHistorialAbierto(false);
    enterBloqueadoRef.current = true;
    setTimeout(() => {
      enterBloqueadoRef.current = false;
    }, 150);
  };

  usePOSAtajos({
    onBuscar: enfocarProducto,
    onPrecio: enfocarPrecio,
    onHistorial: abrirHistorial,
    onArqueo: abrirArqueo,
    onBorrarUltima: borrarUltima,
    onCobrar: cobrar,
    activo: !modalCobroAbierto && !modalArqueoAbierto && !modalHistorialAbierto,
  });

  // -------------------- Handlers de la barra superior --------------------

  const seleccionarProducto = (art) => {
    busqueda.setBusqueda(art.nombre);
    busqueda.cerrar();
    carga.seleccionar(art);
    setTimeout(enfocarPrecio, 50);
  };

  const handleProductoKeyDown = (e) => {
    // ← / → navegan entre campos. ↑ / ↓ los maneja el dropdown.
    if (handleFlechasCampo("producto", e)) return;
    const res = busqueda.handleKeyDown(e);
    if (!res) return;
    if (res.tipo === "seleccionar") {
      seleccionarProducto(res.art);
    } else if (res.tipo === "pasar-a-precio") {
      enfocarPrecio();
    }
  };

  const handleCantidadKeyDown = (e) => {
    if (handleFlechasCampo("cantidad", e)) return;
    if (e.key === "ArrowUp") {
      e.preventDefault();
      e.stopPropagation();
      carga.incrementarCantidad(1);
      return;
    }
    if (e.key === "ArrowDown") {
      e.preventDefault();
      e.stopPropagation();
      carga.incrementarCantidad(-1);
      return;
    }
    if (e.key === "Enter") {
      e.preventDefault();
      e.stopPropagation();
      enfocarPrecio();
    }
  };

  const confirmarCargaYLimpiar = () => {
    console.log(
      "[confirmarCargaYLimpiar] cantidad actual:",
      JSON.stringify(carga.cantidad),
      "precio:",
      JSON.stringify(carga.precio),
    );
    const descripcionTipeada = busqueda.busqueda;
    const r = carga.confirmarCarga(descripcionTipeada);
    if (r.ok) {
      busqueda.limpiar();
    }
    setTimeout(enfocarPrecio, 50);
  };

  const handlePrecioKeyDown = (e) => {
    if (handleFlechasCampo("precio", e)) return;
    if (e.key === "Enter") {
      e.preventDefault();
      if (enterBloqueadoRef.current) return; // Frena el Enter residual
      confirmarCargaYLimpiar();
    }
  };

  // -------------------- Edición inline en la grilla --------------------

  const handleEditarPrecio = (id, valor) => {
    const n = Number(valor);
    if (Number.isFinite(n) && n >= 0 && n <= 999000) {
      pos.editarPrecio(id, n);
    }
  };

  const handleEditarCantidad = (id, valor) => {
    const n = parseInt(valor, 10);
    if (Number.isFinite(n) && n >= 1) {
      pos.editarCantidad(id, n);
    }
  };

  // -------------------- Render --------------------

  return (
    <div className="h-screen w-screen flex flex-col bg-slate-50 text-slate-800 overflow-hidden">
      {/* HEADER */}

// src/context/POSContext.jsx
// src/context/POSContext.jsx
import {
  createContext,
  useContext,
  useReducer,
  useCallback,
  useMemo,
} from "react";

function generarId() {
  return `pos_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`;
}

function redondear2(n) {
  return Math.round((Number(n) || 0) * 100) / 100;
}

function crearFilaVacia() {
  return {
    id: generarId(),
    tipo: "producto",
    sku: null,
    nombre: "",
    cantidad: 1,
    precioUnitario: 0,
    subtotal: 0,
    precioEditado: false,
    stock: null,
    imagen: null,
    vacia: true,
    esNuevo: false,
  };
}

const estadoInicial = {
  items: [crearFilaVacia()],
};

function reducer(state, action) {
  switch (action.type) {
    case "AGREGAR_PRODUCTO": {
      const {
        filaId,
        art,
        cantidad = 1,
        precioForzado = null,
      } = action.payload;
      if (!art) return state;

      const sku = String(art.sku || art.refId || art.id || "").trim();
      if (!sku) return state;

      const precioUnitario =
        precioForzado !== null && precioForzado !== undefined
          ? Number(precioForzado) || 0
          : Number(
              art.costoConDescuento ?? art.costoUnitario ?? art.precio ?? 0,
            );

      const existenteIdx = state.items.findIndex(
        (it) => !it.vacia && it.sku === sku,
      );

      if (existenteIdx >= 0) {
        const nuevosItems = state.items.map((it, i) => {
          if (i === existenteIdx) {
            const nuevaCantidad =
              it.cantidad + Math.max(1, Number(cantidad) || 1);
            return {
              ...it,
              cantidad: nuevaCantidad,
              subtotal: redondear2(nuevaCantidad * it.precioUnitario),
            };
          }
          if (it.id === filaId) return crearFilaVacia();
          return it;
        });
        return { ...state, items: nuevosItems };
      }

      const cant = Math.max(1, Number(cantidad) || 1);
      const nuevosItems = state.items.map((it) => {
        if (it.id !== filaId) return it;
        return {
          id: it.id,
          tipo: "producto",
          sku,
          nombre: String(art.nombre || "Sin nombre"),
          cantidad: cant,
          precioUnitario: redondear2(precioUnitario),
          subtotal: redondear2(cant * precioUnitario),
          precioEditado: false,
          stock: art.stock ?? null,
          imagen: art.imagen || null,
          vacia: false,
          esNuevo: true,
        };
      });

      return { ...state, items: [...nuevosItems, crearFilaVacia()] };
    }

    case "AGREGAR_MANUAL": {
      const { monto = 0, descripcion = "", cantidad = 1 } = action.payload; // <-- Agregamos cantidad
      const precioUnitario = Number(monto) || 0;
      const cant = Math.max(1, Math.floor(Number(cantidad) || 1));

      const nuevoItem = {
        id: generarId(),
        tipo: "manual",
        sku: null,
        nombre:
          descripcion?.trim() ||
          (precioUnitario > 0 ? "Monto manual" : "Art. Kiosco"),
        cantidad: cant, // <-- Ahora usa la cantidad real
        precioUnitario,
        subtotal: redondear2(cant * precioUnitario), // <-- Y calcula el subtotal
        precioEditado: false,
        stock: null,
        imagen: null,
        vacia: false,
        esNuevo: true,
      };

      const filasReales = state.items.filter((it) => !it.vacia);
      const filaVacia = state.items.find((it) => it.vacia) || crearFilaVacia();

      return {
        ...state,
        items: [...filasReales, nuevoItem, filaVacia],
      };
    }

    case "EDITAR_PRECIO": {
      const { id, nuevoPrecio } = action.payload;
      const precio = Number(nuevoPrecio);
      if (!Number.isFinite(precio) || precio < 0) return state;

      return {
        ...state,
        items: state.items.map((it) =>
          it.id === id
            ? {
                ...it,
                precioUnitario: redondear2(precio),
                subtotal: redondear2(precio * it.cantidad),
                precioEditado: true,
                esNuevo: false,
              }
            : it,
        ),
      };
    }

    case "EDITAR_CANTIDAD": {
      const { id, nuevaCantidad } = action.payload;
      const cant = Math.max(1, Math.floor(Number(nuevaCantidad) || 1));

      return {
        ...state,
        items: state.items.map((it) =>
          it.id === id
            ? {
                ...it,
                cantidad: cant,
                subtotal: redondear2(cant * it.precioUnitario),
              }
            : it,
        ),
      };
    }

    case "EDITAR_NOMBRE": {
      const { id, nuevoNombre } = action.payload;
      return {
        ...state,
        items: state.items.map((it) =>
          it.id === id
            ? { ...it, nombre: String(nuevoNombre || "").trim() || it.nombre }
            : it,
        ),
      };
    }

    case "QUITAR": {
      const { id } = action.payload;
      const sinFila = state.items.filter((it) => it.id !== id);
      return {
        ...state,
        items: sinFila.length === 0 ? [crearFilaVacia()] : sinFila,
      };
    }

    case "QUITAR_ULTIMA": {
      const filasReales = state.items.filter((it) => !it.vacia);
      if (filasReales.length === 0) return state;

      const sinUltima = filasReales.slice(0, -1);
      return {
        ...state,
        items: [...sinUltima, crearFilaVacia()],
      };
    }

    case "VACIAR":
      return { ...state, items: [crearFilaVacia()] };

    case "LIMPIAR_NUEVO": {
      const { id } = action.payload;
      return {
        ...state,
        items: state.items.map((it) =>
          it.id === id ? { ...it, esNuevo: false } : it,
        ),
      };
    }

    default:
      return state;
  }
}

const POSContext = createContext(null);

export function POSProvider({ children }) {
  const [state, dispatch] = useReducer(reducer, estadoInicial);

  const agregarProducto = useCallback(
    (filaId, art, cantidad = 1, precioForzado = null) => {
      dispatch({
        type: "AGREGAR_PRODUCTO",
        payload: { filaId, art, cantidad, precioForzado },
      });
    },
    [],
  );

  const agregarManual = useCallback(
    (monto = 0, descripcion = "", cantidad = 1) => {
      dispatch({
        type: "AGREGAR_MANUAL",
        payload: { monto, descripcion, cantidad },
      });
    },
    [],
  );

  const editarPrecio = useCallback((id, nuevoPrecio) => {
    dispatch({ type: "EDITAR_PRECIO", payload: { id, nuevoPrecio } });
  }, []);

  const editarCantidad = useCallback((id, nuevaCantidad) => {
    dispatch({ type: "EDITAR_CANTIDAD", payload: { id, nuevaCantidad } });
  }, []);

  const editarNombre = useCallback((id, nuevoNombre) => {
    dispatch({ type: "EDITAR_NOMBRE", payload: { id, nuevoNombre } });
  }, []);

  const quitarItem = useCallback((id) => {
    dispatch({ type: "QUITAR", payload: { id } });
  }, []);

  const quitarUltima = useCallback(() => {
    dispatch({ type: "QUITAR_ULTIMA" });
  }, []);

  const vaciar = useCallback(() => dispatch({ type: "VACIAR" }), []);

  const limpiarNuevo = useCallback((id) => {
    dispatch({ type: "LIMPIAR_NUEVO", payload: { id } });
  }, []);

  const totales = useMemo(() => {
    const filas = state.items.filter((it) => !it.vacia);
    const totalUnidades = filas.reduce((s, it) => s + it.cantidad, 0);
    const total = redondear2(filas.reduce((s, it) => s + it.subtotal, 0));
    return {
      cantidadItems: filas.length,
      totalUnidades,
      total,
    };
  }, [state.items]);

  /**
   * Construye el ticket con el CONTRATO que espera el backend:
   *
   *   { id: null, fecha, pagos[], metodoPago, total, totalPagado, vuelto, items[] }
   *
   * IMPORTANTE:
   *   - `id` se manda null. El backend lo asigna atómicamente al
   *     guardar (T-00001, T-00002, ...). No confiamos en IDs del front.
   *   - `fecha` es ISO local; el backend usa `fechaArchivo` (YYYY-MM-DD)
   *     para decidir el archivo del día si viene, si no usa la fecha.
   */
  const construirTicket = useCallback(
    (pagos = []) => {
      const pagosNormalizados = (Array.isArray(pagos) ? pagos : []).map(
        (p) => ({
          metodo: String(p?.metodo || "Efectivo"),
          monto: redondear2(p?.monto),
        }),
      );

      const totalPagado = redondear2(
        pagosNormalizados.reduce((s, p) => s + p.monto, 0),
      );
      const vuelto = redondear2(Math.max(0, totalPagado - totales.total));

      const metodoPago =
        pagosNormalizados.length === 0
          ? "Efectivo"
          : pagosNormalizados.length === 1
            ? pagosNormalizados[0].metodo
            : "Mixto";

      const ahora = new Date();

      return {
        id: null, // ← el backend lo asigna (T-00001, T-00002, ...)
        fecha: ahora.toISOString(),
        // YYYY-MM-DD en hora local, para que el backend guarde en el
        // archivo del día correcto (no se desfasa por timezone).
        fechaArchivo: ahora.toISOString().slice(0, 10),
        pagos: pagosNormalizados,
        metodoPago,
        total: totales.total,
        totalPagado,
        vuelto,
        items: state.items
          .filter((it) => !it.vacia)
          .map((it) => ({
            tipo: it.tipo,
            sku: it.sku,
            nombre: it.nombre,
            cantidad: it.cantidad,
            precioUnitario: it.precioUnitario,
            subtotal: it.subtotal,
            precioEditado: it.precioEditado,
          })),
      };
    },
    [state.items, totales.total],
  );

  const value = {
    items: state.items,
    itemsReales: state.items.filter((it) => !it.vacia),
    totales,
    agregarProducto,
    agregarManual,
    editarPrecio,
    editarCantidad,
    editarNombre,
    quitarItem,
    quitarUltima,
    vaciar,
    limpiarNuevo,
    construirTicket,
  };

  return <POSContext.Provider value={value}>{children}</POSContext.Provider>;
}

export function usePOS() {
  const ctx = useContext(POSContext);
  if (!ctx) throw new Error("usePOS debe usarse dentro de <POSProvider>");
  return ctx;
}

// frontend/src/hooks/usePOSAtajos.js
import { useEffect } from "react";

/**
 * ATAJOS DE TECLADO DEL POS
 * Registra los atajos globales en fase CAPTURE (para que el navegador
 * no se robe F5/F10/etc).
 *
 * @param {object} opts
 * @param {Function} opts.onBuscar       F1
 * @param {Function} opts.onPrecio       F2
 * @param {Function} opts.onHistorial    F3
 * @param {Function} opts.onArqueo       F4
 * @param {Function} opts.onBorrarUltima F9
 * @param {Function} opts.onCobrar       F10
 * @param {boolean}  opts.activo         false → no registra nada
 */
export function usePOSAtajos({
  onBuscar,
  onPrecio,
  onHistorial,
  onArqueo,
  onBorrarUltima,
  onCobrar,
  activo = true,
}) {
  useEffect(() => {
    if (!activo) return;

    const handler = (e) => {
      switch (e.key) {
        case "F1":
          e.preventDefault();
          onBuscar?.();
          break;
        case "F2":
          e.preventDefault();
          onPrecio?.();
          break;
        case "F3":
          e.preventDefault();
          onHistorial?.();
          break;
        case "F4":
          e.preventDefault();
          onArqueo?.();
          break;
        case "F9":
          e.preventDefault();
          onBorrarUltima?.();
          break;
        case "F10":
          e.preventDefault();
          onCobrar?.();
          break;
        default:
          break;
      }
    };

    window.addEventListener("keydown", handler, true);
    return () => window.removeEventListener("keydown", handler, true);
  }, [
    onBuscar,
    onPrecio,
    onHistorial,
    onArqueo,
    onBorrarUltima,
    onCobrar,
    activo,
  ]);

  return {
    buscar: () => onBuscar?.(),
    precio: () => onPrecio?.(),
    historial: () => onHistorial?.(),
    arqueo: () => onArqueo?.(),
    borrarUltima: () => onBorrarUltima?.(),
    cobrar: () => onCobrar?.(),
  };
}```

### `frontend/src/hooks/usePOSBusqueda.js`
```js
// frontend/src/hooks/usePOSBusqueda.js
import { useState, useMemo, useCallback } from "react";

/**
 * ============================================================
 * BÚSQUEDA / DROPDOWN DE PRODUCTOS DEL POS
 * ============================================================
 *
 * Maneja:
 *   - texto del input
 *   - apertura/cierre del dropdown
 *   - índice seleccionado (flechas ↑↓)
 *   - filtrado en memoria (nombre / marca / sku)
 *   - selección con Enter
 *
 * @param {Array} catalogo  Productos cargados del proveedor actual
 * @param {number} maxResultados
 */
export function usePOSBusqueda(catalogo, maxResultados = 8) {
  const [busqueda, setBusqueda] = useState("");
  const [dropAbierto, setDropAbierto] = useState(false);
  const [indiceDrop, setIndiceDrop] = useState(0);

  const resultados = useMemo(() => {
    const t = busqueda.trim().toLowerCase();
    if (!t) return [];
    return (catalogo || [])
      .filter((a) => {
        const blob =
          `${a.nombre || ""} ${a.marca || ""} ${a.sku || ""}`.toLowerCase();
        return blob.includes(t);
      })
      .slice(0, maxResultados);
  }, [busqueda, catalogo, maxResultados]);

  /**
   * Busca por coincidencia EXACTA de sku/refId (para escaneo).
   * Devuelve el producto o null.
   */
  const buscarPorCodigo = useCallback(
    (codigo) => {
      const t = String(codigo || "")
        .trim()
        .toLowerCase();
      if (!t) return null;
      return (
        (catalogo || []).find(
          (a) =>
            String(a.sku || "").toLowerCase() === t ||
            String(a.refId || "").toLowerCase() === t,
        ) || null
      );
    },
    [catalogo],
  );

  const handleChange = useCallback((valor) => {
    setBusqueda(valor);
    setDropAbierto(valor.length > 0);
    setIndiceDrop(0);
  }, []);

  const cerrar = useCallback(() => {
    setDropAbierto(false);
  }, []);

  const limpiar = useCallback(() => {
    setBusqueda("");
    setDropAbierto(false);
    setIndiceDrop(0);
  }, []);

  /**
   * Maneja teclas del input del dropdown.
   * Devuelve la acción a ejecutar (o null si no aplica).
   *
   * - ArrowDown / ArrowUp → mueve índice
   * - Enter con dropdown abierto → selecciona el del índice
   * - Escape → cierra
   * - Enter con dropdown cerrado → intenta buscar por código exacto
   *
   * El caller decide qué hacer con el producto devuelto.
   */
  const handleKeyDown = useCallback(
    (e) => {
      if (dropAbierto && resultados.length > 0) {
        if (e.key === "ArrowDown") {
          e.preventDefault();
          setIndiceDrop((prev) => (prev + 1) % resultados.length);
          return { tipo: "navegar" };
        }
        if (e.key === "ArrowUp") {
          e.preventDefault();
          setIndiceDrop(
            (prev) => (prev - 1 + resultados.length) % resultados.length,
          );
          return { tipo: "navegar" };
        }
        if (e.key === "Enter") {
          e.preventDefault();
          const art = resultados[indiceDrop] || resultados[0]; // <-- Fallback al primero
          return art ? { tipo: "seleccionar", art } : { tipo: "vacio" };
        }
        if (e.key === "Escape") {
          e.preventDefault();
          setDropAbierto(false);
          return { tipo: "cerrar" };
        }
      } else if (e.key === "Enter") {
        e.preventDefault();
        const art = buscarPorCodigo(busqueda);
        return art ? { tipo: "seleccionar", art } : { tipo: "pasar-a-precio" };
      }
      return null;
    },
    [dropAbierto, resultados, indiceDrop, buscarPorCodigo, busqueda],
  );

  return {
    // estado
    busqueda,
    dropAbierto,
    indiceDrop,
    resultados,
    // acciones
    setBusqueda,
    setIndiceDrop,
    handleChange,
    handleKeyDown,
    cerrar,
    limpiar,
    buscarPorCodigo,
  };
}
```

### `frontend/src/hooks/usePOSCarga.js`
```js
// frontend/src/hooks/usePOSCarga.js
import { useState, useCallback } from "react";

/**
 * Tope de precio. Si un EAN escaneado por error cae acá, se rechaza
 * (es el bug clásico de "pistola de códigos en el input de precio").
 */
export const TOPE_PRECIO = 999000;

/**
 * Tope de cantidad. Evita que un "9999999" tipeado por error rompa
 * los totales del ticket.
 */
export const TOPE_CANTIDAD = 9999;

/**
 * ============================================================
 * CARGA DE PRODUCTO EN LA BARRA SUPERIOR
 * ============================================================
 *
 * Maneja el estado de la barra superior:
 *   - artSeleccionado (producto del catálogo, o null → manual)
 *   - cantidad (STRING, para permitir borrar y tipear libre)
 *   - precio   (STRING, idem)
 *
 * `cantidad` es string a propósito: si fuera número, borrar el input
 * lo forzaría a 1 y no se podría tipear "2" (el clásico bug del
 * input numérico controlado). Se parsea recién en `confirmarCarga`.
 *
 * @param {object} opts
 * @param {Function} opts.onProducto  (art, cantidad, precio) → void
 * @param {Function} opts.onManual    (precio, descripcion) → void
 */
export function usePOSCarga({ onProducto, onManual }) {
  const [artSeleccionado, setArtSeleccionado] = useState(null);
  const [cantidad, setCantidad] = useState("1"); // string
  const [precio, setPrecio] = useState("");
  const [error, setError] = useState(null);

  /**
   * Selecciona un producto del catálogo.
   * Precarga su costo como precio por defecto y resetea cantidad a 1.
   */
  const seleccionar = useCallback((art) => {
    if (!art) return;
    setArtSeleccionado(art);

    const costo = Number(
      art.costoConDescuento ?? art.costoUnitario ?? art.precio ?? 0,
    );
    setPrecio(costo > 0 ? costo.toFixed(2) : "");
    setCantidad("1");
    setError(null);
  }, []);

  /**
   * Deselecciona el producto del catálogo SIN tocar cantidad ni precio.
   * A diferencia de `seleccionar(null)` (que es no-op por el `if (!art) return`),
   * esto sí limpia `artSeleccionado` para que el próximo Enter en precio
   * agregue un ítem manual en vez del producto viejo.
   */
  const deseleccionar = useCallback(() => setArtSeleccionado(null), []);

  /**
   * Acepta el string tal cual, permitiendo "" (input vacío mientras
   * el usuario tipea) y dígitos. Ignora cualquier otra cosa.
   * NO clampa acá: el clamp a 1 se hace en `confirmarCarga`, para no
   * romper el tipeo intermedio.
   */
  const cambiarCantidad = useCallback((nueva) => {
    console.log("[cambiarCantidad] recibe:", JSON.stringify(nueva));
    const s = String(nueva);
    if (s === "" || /^\d+$/.test(s)) {
      setCantidad(s);
      console.log("[cambiarCantidad] setea a:", JSON.stringify(s));
    }
  }, []);

  /**
   * Suma/resta al valor actual. Si el string está vacío, parte de 1.
   * Se usa desde ↑ / ↓ en el input de cantidad.
   */
  const incrementarCantidad = useCallback((delta) => {
    setCantidad((prev) => {
      const base = prev === "" ? 1 : Number(prev) || 1;
      const n = Math.max(1, base + delta);
      return String(n);
    });
  }, []);

  const cambiarPrecio = useCallback((nuevo) => {
    setPrecio(nuevo);
    setError(null);
  }, []);

  const limpiar = useCallback(() => {
    console.log('[limpiar] llamado, resetea cantidad a "1"');
    setArtSeleccionado(null);
    setCantidad("1");
    setPrecio("");
    setError(null);
  }, []);

  /**
   * Confirma la carga en la grilla.
   *
   * Reglas:
   *  - precio > 0
   *  - precio <= TOPE_PRECIO
   *  - cantidad entre 1 y TOPE_CANTIDAD (clampea a 1 si vacío/0)
   *  - si hay art seleccionado → onProducto(art, cantidad, precio)
   *  - si no → onManual(precio, descripcionTipeada || "Art. Kiosco")
   *
   * Devuelve { ok: boolean, error?: string }.
   */
  const confirmarCarga = useCallback(
    (descripcionTipeada = "") => {
      const precioNum = Number(precio) || 0;
      const cantNum = Math.max(1, Math.floor(Number(cantidad) || 0));

      if (precioNum <= 0) {
        setError("El precio debe ser mayor a 0");
        return { ok: false, error: "precio<=0" };
      }

      if (precioNum > TOPE_PRECIO) {
        setError(
          `El precio no puede superar los $${TOPE_PRECIO.toLocaleString(
            "es-AR",
          )} (¿Escaneaste un código por error?)`,
        );
        return { ok: false, error: "precio>tope" };
      }

      if (cantNum > TOPE_CANTIDAD) {
        setError(`La cantidad no puede superar ${TOPE_CANTIDAD}`);
        return { ok: false, error: "cantidad>tope" };
      }

      if (artSeleccionado) {
        onProducto?.(artSeleccionado, cantNum, precioNum);
      } else {
        const desc = String(descripcionTipeada || "").trim() || "Art. Kiosco";
        onManual?.(precioNum, desc, cantNum); // <-- Agregamos cantNum como 3er parámetro
      }

      limpiar();
      return { ok: true };
    },
    [precio, cantidad, artSeleccionado, onProducto, onManual, limpiar],
  );

  return {
    // estado
    artSeleccionado,
    cantidad,
    precio,
    error,
    // acciones
    seleccionar,
    deseleccionar,
    cambiarCantidad,
    incrementarCantidad,
    cambiarPrecio,
    limpiar,
    confirmarCarga,
    setError,
  };
}

// frontend/src/services/posApi.js
import { api } from "../api/client.js";

/**
 * Wrappers sobre `api.*` con manejo de error uniforme y normalización.
 * No tiene estado. No toca React.
 */

// ==================== ARQUEO ====================

export async function cerrarCaja({ montoInicial, efectivoDeclarado }) {
  const inicial = Number(montoInicial);
  const declarado = Number(efectivoDeclarado);

  if (!Number.isFinite(inicial) || inicial < 0) {
    throw new Error("Monto inicial inválido");
  }
  if (!Number.isFinite(declarado) || declarado < 0) {
    throw new Error("Efectivo declarado inválido");
  }

  const data = await api.posCerrarCaja(inicial, declarado);

  if (!data?.success || !data.resumen) {
    throw new Error(data?.error || "El backend no devolvió un resumen válido");
  }

  return data.resumen;
}

// ==================== REIMPRESIÓN ====================

export async function reimprimirTicket(ticket) {
  const data = await api.posImprimir(ticket);
  if (!data?.success) {
    throw new Error(data?.error || "No se pudo reimprimir");
  }
  return data;
}

// ==================== VENTAS POR FECHA ====================

/**
 * Trae las ventas de una fecha (YYYY-MM-DD). Si `fecha` es falsy, hoy.
 * Devuelve un array de ventas ordenado de más nueva a más vieja.
 * Si el archivo no existe, el backend devuelve [] (no error).
 */
export async function getVentasPorFecha(fecha) {
  const data = await api.posVentasHoy(fecha);
  if (!data?.success || !Array.isArray(data.ventas)) {
    throw new Error(data?.error || "Respuesta inválida del backend");
  }
  return [...data.ventas].reverse();
}

// ==================== RESUMEN DEL DÍA ====================

/**
 * Resumen agregado del día (server-side). Devuelve:
 * { fecha, cantidadVentas, totalFacturado, porMetodo: { Efectivo: N, ... } }
 */
export async function getResumenDia(fecha) {
  const data = await api.posResumenDia(fecha);
  if (!data?.success) {
    throw new Error(data?.error || "No se pudo obtener el resumen del día");
  }
  return {
    fecha: data.fecha,
    cantidadVentas: data.cantidadVentas ?? 0,
    totalFacturado: data.totalFacturado ?? 0,
    porMetodo: data.porMetodo ?? {},
  };
}

// ==================== SIGUIENTE TICKET ====================

/**
 * Devuelve el próximo ID de ticket (formateado, p.ej. "T-00042") SIN
 * reservarlo. Es informativo para la UI: la reserva real ocurre al
 * guardar la venta en POST /api/pos/venta.
 */
export async function getSiguienteTicket() {
  const data = await api.posSiguienteTicket();
  if (!data?.success || !data.id) {
    throw new Error(data?.error || "No se pudo obtener el próximo ticket");
  }
  return data.id;
}```

### `start.sh`
```sh
#!/usr/bin/env bash
# start.sh — Levanta backend y frontend del POS Dulce Reino en WSL.
# Uso: ./start.sh

set -uo pipefail

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
BACKEND_DIR="$SCRIPT_DIR/backend"
FRONT_DIR="$SCRIPT_DIR/frontend"

echo "🚀 Levantando POS Dulce Reino..."
echo "   Backend:  $BACKEND_DIR  →  http://localhost:4000"
echo "   Frontend: $FRONT_DIR    →  http://localhost:5173"
echo ""

# Cargar pnpm si no está en el PATH (pasa cuando se ejecuta desde
# el acceso directo de Windows, que no carga .bashrc).
if ! command -v pnpm >/dev/null 2>&1; then
  [ -d "$HOME/.local/share/pnpm" ] && export PATH="$HOME/.local/share/pnpm:$PATH"
  [ -s "$HOME/.nvm/nvm.sh" ] && source "$HOME/.nvm/nvm.sh" >/dev/null 2>&1
fi

if ! command -v pnpm >/dev/null 2>&1; then
  echo "❌ ERROR: pnpm no está en el PATH."
  echo "   Instalalo con: npm install -g pnpm"
  exit 1
fi

# Verificar que existan las dependencias instaladas en cada subproyecto.
# Si no, avisamos en vez de fallar con un error críptico de Vite/nodemon.
if [ ! -d "$BACKEND_DIR/node_modules" ]; then
  echo "⚠️  Falta backend/node_modules. Corré: (cd backend && pnpm install)"
  exit 1
fi
if [ ! -d "$FRONT_DIR/node_modules" ]; then
  echo "⚠️  Falta frontend/node_modules. Corré: (cd frontend && pnpm install)"
  exit 1
fi

echo "✅ pnpm $(pnpm --version)"
echo ""

# Trap para matar los procesos al salir (Ctrl+C, cierre de terminal, etc.)
cleanup() {
  echo ""
  echo "🛑 Deteniendo..."
  kill $(jobs -p) 2>/dev/null
  wait 2>/dev/null
  exit 0
}
trap cleanup EXIT INT TERM

# Backend
( cd "$BACKEND_DIR" && pnpm dev ) &
BACKEND_PID=$!
echo "✅ Backend lanzado (PID: $BACKEND_PID)"

# Frontend
( cd "$FRONT_DIR" && pnpm dev ) &
FRONT_PID=$!
echo "✅ Frontend lanzado (PID: $FRONT_PID)"

echo ""
echo "⏳ Esperando... (Ctrl+C para detener todo)"
echo ""

# Si cualquiera de los dos muere, matamos el otro y salimos.
# Esto evita el caso feo de "el backend crasheó pero el front sigue vivo".
wait -n
echo "⚠️  Uno de los procesos terminó. Cerrando el otro..."
cleanup```
